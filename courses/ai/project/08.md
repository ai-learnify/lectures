### ლექციაზე
- **ერთად წერთ:** `Program.cs`-ის რეგისტრაცია (`AddHttpClient("gemini")` → `AddStandardResilienceHandler` → `AddChatClient(sp => …)`, `Attempts = 1`) და `FallbackChatClient`.
- **ჩასვამ და ახსნი:** `ChaosHandler` (ცდისთვის) და ახალი `TokenCounterChatClient` (ბაზაში).
- **წინასწარ:** ყველაფერი შენთან მუშაობს; `ChaosHandler`-ით ერთხელ გაშვებული და ნანახი, რას წერს ლოგი; NuGet-ის ვერსია ჩაწერილი — სტუდენტებს იგივე სჭირდებათ.

### რიგი
- **HttpClient:** პირველი დამატებული — გარე. resilience გარეთ, `ChaosHandler` შიგნით (ქსელთან) → retry ხედავს chaos-ის 503-ს.
- **Chat pipeline:** `FallbackChatClient` ყველაზე შიგნითაა (Gemini-სთან): retry-ის შემდეგაც თუ ჩავარდა, სხვა მოდელს იძახებს — და `TokenCounter`-იც და `UseLogging`-იც ამ გამოძახებას ხედავს.
- **`TokenCounterChatClient` Singleton pipeline-შია**, `DataContext` კი Scoped: ამიტომ `IServiceScopeFactory`-ით საკუთარ scope-ს ქმნის (captive dependency-ის გარეშე).

### შემოწმება
- `ChaosHandler`-ის ხაზი ჩართე → `/chat` → ლოგში ორი 503 და მერე 200 (retry).
- `Gemini:Model` = `gemini-0` → 404, **ერთი** მოთხოვნა: retry მუდმივ შეცდომაზე არ მეორდება.
- დღის ხარჯი:
```sql
SELECT "Model", SUM("InputTokens") AS input, SUM("OutputTokens") AS output
FROM "AiUsage"
WHERE "At" >= CURRENT_DATE
GROUP BY "Model";
```
ფასი ლექცია 1-ის ფორმულით: input × input-ის ფასი + output × output-ის ფასი (1M ტოკენზე). `Model` სვეტით ჩანს, რამდენჯერ ჩაერთო fallback.

```csharp +Program.cs
using Google.GenAI;
using Google.GenAI.Types;
using Microsoft.Extensions.AI;

var builder = WebApplication.CreateBuilder(args);

// ---------- AI ----------
string apiKey = builder.Configuration["Gemini:ApiKey"]
    ?? throw new InvalidOperationException("Gemini:ApiKey is missing");
string model = builder.Configuration["Gemini:Model"]
    ?? throw new InvalidOperationException("Gemini:Model is missing");

builder.Services.AddTransient<ChaosHandler>();
builder.Services.AddHttpClient("gemini")
    .AddStandardResilienceHandler(o =>
    {
        o.AttemptTimeout.Timeout = TimeSpan.FromSeconds(40);   // an LLM is slow
        o.TotalRequestTimeout.Timeout = TimeSpan.FromSeconds(90);
        o.CircuitBreaker.SamplingDuration = TimeSpan.FromSeconds(90);
    });
    // .AddHttpMessageHandler<ChaosHandler>();   // ⚠ only for testing: inside, close to the network

IChatClient fallback = new Client(apiKey: apiKey)
    .AsIChatClient(builder.Configuration["Gemini:FallbackModel"]!);

builder.Services.AddChatClient(sp =>
    {
        IHttpClientFactory http = sp.GetRequiredService<IHttpClientFactory>();
        Client gemini = new(
            apiKey: apiKey,
            httpOptions: new HttpOptions { RetryOptions = new HttpRetryOptions { Attempts = 1 } },
            clientOptions: new ClientOptions { HttpClientFactory = () => http.CreateClient("gemini") });
        return gemini.AsIChatClient(model);
    })
    .Use(inner => new InputGuardChatClient(inner))          // 1 — stops earliest
    .UseFunctionInvocation()                                // 2 — the tool loop
    .Use((inner, sp) => new TokenCounterChatClient(         // 3 — every Gemini call → database
        inner, sp.GetRequiredService<IServiceScopeFactory>()))
    .UseLogging()                                           // 4 — what goes to Gemini
    .Use((inner, sp) => new FallbackChatClient(             // 5 — failed → another model
        inner, fallback, sp.GetRequiredService<ILogger<FallbackChatClient>>()));

builder.Services.AddScoped<RestaurantTools>();
builder.Services.AddScoped<ChatHistoryStore>();
// ------------------------

builder.Services.AddControllers();
// ... the rest — as it was
```

```csharp Ai/ChaosHandler.cs
using System.Net;

// only for testing: the first two of every 3 requests → 503
public class ChaosHandler : DelegatingHandler
{
    private int _count;

    protected override Task<HttpResponseMessage> SendAsync(
        HttpRequestMessage request, CancellationToken ct)
    {
        if (Interlocked.Increment(ref _count) % 3 != 0)
            return Task.FromResult(
                new HttpResponseMessage(HttpStatusCode.ServiceUnavailable));

        return base.SendAsync(request, ct);
    }
}
```

```csharp Ai/FallbackChatClient.cs
using Google.GenAI;
using Microsoft.Extensions.AI;
using Polly.Timeout;

public class FallbackChatClient : DelegatingChatClient
{
    private readonly IChatClient _fallback;
    private readonly ILogger<FallbackChatClient> _logger;

    public FallbackChatClient(IChatClient inner, IChatClient fallback,
        ILogger<FallbackChatClient> logger) : base(inner)
    {
        _fallback = fallback;
        _logger = logger;
    }

    public override async Task<ChatResponse> GetResponseAsync(
        IEnumerable<ChatMessage> messages, ChatOptions? options = null,
        CancellationToken cancellationToken = default)
    {
        try
        {
            return await base.GetResponseAsync(messages, options, cancellationToken);
        }
        catch (Exception ex) when (IsTransient(ex))
        {
            _logger.LogWarning("🛟 {Error} — fallback model", ex.Message);
            return await _fallback.GetResponseAsync(messages, options, cancellationToken);
        }
    }

    private static bool IsTransient(Exception ex) => ex switch
    {
        ApiException api => api.StatusCode is 429 or >= 500,
        HttpRequestException or TimeoutRejectedException => true,
        _ => false
    };
}
```

```csharp Ai/TokenCounterChatClient.cs
using Microsoft.Extensions.AI;

public class TokenCounterChatClient : DelegatingChatClient
{
    private readonly IServiceScopeFactory _scopes;

    public TokenCounterChatClient(IChatClient inner, IServiceScopeFactory scopes) : base(inner)
    {
        _scopes = scopes;
    }

    public override async Task<ChatResponse> GetResponseAsync(
        IEnumerable<ChatMessage> messages, ChatOptions? options = null,
        CancellationToken cancellationToken = default)
    {
        ChatResponse response =
            await base.GetResponseAsync(messages, options, cancellationToken);

        // a Scoped DbContext in a Singleton layer → its own scope
        using IServiceScope scope = _scopes.CreateScope();
        DataContext db = scope.ServiceProvider.GetRequiredService<DataContext>();

        db.AiUsage.Add(new AiUsageRecord
        {
            Id = Guid.NewGuid(),
            At = DateTime.UtcNow,
            Model = response.ModelId ?? "?",
            InputTokens = response.Usage?.InputTokenCount ?? 0,
            OutputTokens = (response.Usage?.OutputTokenCount ?? 0)
                         + (response.Usage?.ReasoningTokenCount ?? 0)
        });
        await db.SaveChangesAsync(cancellationToken);

        return response;
    }
}
```

### მეგობრული შეცდომა კლიენტს
თუ fallback-იც ჩავარდა, კლიენტს 500-ის ნაცვლად 503 + `Retry-After`:

```csharp Ai/AiExceptionHandler.cs — ⭐
public class AiExceptionHandler : IExceptionHandler
{
    public async ValueTask<bool> TryHandleAsync(HttpContext ctx, Exception ex, CancellationToken ct)
    {
        if (ex is not ApiException api) return false;
        ctx.Response.StatusCode = StatusCodes.Status503ServiceUnavailable;
        ctx.Response.Headers.RetryAfter = "30";
        await ctx.Response.WriteAsJsonAsync(new ProblemDetails
        {
            Title = "The assistant is temporarily unavailable",
            Detail = "Try again in a little while.",
            Status = 503
        }, ct);
        return true;
    }
}
// Program.cs: builder.Services.AddExceptionHandler<AiExceptionHandler>(); app.UseExceptionHandler();
```
