- **ერთად წერთ:** `ChatMessageRecord`, `ChatRequest` და ახალი `/chat` (ჩატვირთე → `[SystemPrompt(), .. history, question]` → შეინახე `[question, .. response.Messages]`), შემდეგ reducer.
- **ჩასვამ და ახსნი:** `ChatHistoryStore` (გრძელია) და migration.

```csharp Ai/ChatHistoryStore.cs
using System.Text.Json;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.AI;

public class ChatHistoryStore
{
    // jsonb reorders keys — "$type" is no longer first
    private static readonly JsonSerializerOptions Json =
        new(AIJsonUtilities.DefaultOptions) { AllowOutOfOrderMetadataProperties = true };

    private readonly DataContext _db;
    public ChatHistoryStore(DataContext db) { _db = db; }

    public async Task<List<ChatMessage>> LoadAsync(Guid conversationId, CancellationToken ct)
    {
        List<string> rows = await _db.ChatMessages
            .Where(m => m.ConversationId == conversationId)
            .OrderBy(m => m.Position)
            .Select(m => m.Json)
            .ToListAsync(ct);

        return rows.Select(j => JsonSerializer.Deserialize<ChatMessage>(j, Json)!).ToList();
    }

    public async Task AppendAsync(Guid conversationId, IEnumerable<ChatMessage> messages, CancellationToken ct)
    {
        int position = await _db.ChatMessages.CountAsync(m => m.ConversationId == conversationId, ct);
        foreach (ChatMessage message in messages)
        {
            _db.ChatMessages.Add(new ChatMessageRecord
            {
                Id = Guid.NewGuid(),
                ConversationId = conversationId,
                Position = position++,
                Role = message.Role.Value,
                Json = JsonSerializer.Serialize(message, Json),
                CreatedAt = DateTime.UtcNow
            });
        }
        await _db.SaveChangesAsync(ct);
    }
}
```

```csharp Controllers/AiLabController.cs
using System.ComponentModel;
using System.Diagnostics;
using System.Runtime.CompilerServices;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.AI;

[ApiController]
[Route("api/ai-lab")]
public class AiLabController : ControllerBase
{
    private const string RestaurantName = "Supra";
    private const string OpenHours = "12:00–23:00";

#pragma warning disable MEAI001   // reducers are still experimental
    private static readonly IChatReducer Reducer = new MessageCountingChatReducer(10);
#pragma warning restore MEAI001

    private readonly IChatClient _chat;
    private readonly RestaurantTools _tools;
    private readonly ChatHistoryStore _history;

    public AiLabController(IChatClient chat, RestaurantTools tools, ChatHistoryStore history)
    {
        _chat = chat;
        _tools = tools;
        _history = history;
    }

    [HttpPost("ask")]
    public async Task<IActionResult> Ask(AskRequest request)
    {
        var options = new ChatOptions
        {
            Temperature = request.Temperature,
            MaxOutputTokens = request.MaxOutputTokens
        };

        ChatResponse response =
            await _chat.GetResponseAsync(request.Prompt, options);
        return Ok(AskResponse.From(response));
    }

    [HttpPost("assistant")]
    public async Task<IActionResult> Assistant(AskRequest request)
    {
        List<ChatMessage> messages =
        [
            new(ChatRole.System, $"""
                You are the assistant of the restaurant "{RestaurantName}".
                You answer only about the menu, bookings and hours ({OpenHours}).
                Answer: at most 3 sentences, in the language the user writes in.
                On any other topic you politely decline and say what you can help with.
                You don't invent prices or ingredients — you say they are on the menu.
                """),
            new(ChatRole.User, request.Prompt)
        ];

        ChatResponse response = await _chat.GetResponseAsync(messages);
        return Ok(AskResponse.From(response));
    }

    public enum Sentiment { Positive, Negative, Mixed }

    [HttpPost("review-sentiment")]
    public async Task<IActionResult> ReviewSentiment(AskRequest request)
    {
        List<ChatMessage> messages =
        [
            new(ChatRole.System, "Classify the restaurant review."),
            new(ChatRole.User, request.Prompt)
        ];

        ChatResponse<Sentiment> response =
            await _chat.GetResponseAsync<Sentiment>(messages);

        Sentiment sentiment = response.Result;
        if (sentiment == Sentiment.Negative) { /* notify the manager */ }

        return Ok(sentiment.ToString());
    }

    public record BookingRequest(
        [property: Description("Booking date and time")] DateTime? At,
        [property: Description("Number of guests")] int? Guests);

    [HttpPost("parse-booking")]
    public async Task<IActionResult> ParseBooking(AskRequest request)
    {
        List<ChatMessage> messages =
        [
            new(ChatRole.System, $"""
                Extract the booking details from the user's text.
                Today is {DateTime.Now:yyyy-MM-dd, dddd}.
                Leave null whatever is not in the text.
                """),
            new(ChatRole.User, request.Prompt)
        ];

        ChatResponse<BookingRequest> response =
            await _chat.GetResponseAsync<BookingRequest>(messages);
        return Ok(response.Result);
    }

    public record ChatRequest(Guid? ConversationId, string Prompt);

    [HttpPost("chat")]
    public async Task<IActionResult> Chat(ChatRequest request, CancellationToken ct)
    {
        Guid conversationId = request.ConversationId ?? Guid.NewGuid();

        List<ChatMessage> history = await _history.LoadAsync(conversationId, ct);
        IEnumerable<ChatMessage> recent = await Reducer.ReduceAsync(history, ct);

        ChatMessage question = new(ChatRole.User, request.Prompt);
        List<ChatMessage> messages = [SystemPrompt(), .. recent, question];

        ChatResponse response =
            await _chat.GetResponseAsync(messages, ChatToolOptions(), ct);

        await _history.AppendAsync(conversationId, [question, .. response.Messages], ct);

        return Ok(new { conversationId, response.Text, response.Usage?.InputTokenCount });
    }

    [HttpPost("chat/chunks")]
    public async Task<IActionResult> ChatChunks(AskRequest request)
    {
        Stopwatch sw = Stopwatch.StartNew();
        List<object> chunks = [];

        await foreach (ChatResponseUpdate update in
            _chat.GetStreamingResponseAsync(request.Prompt))
        {
            chunks.Add(new { Ms = sw.ElapsedMilliseconds, update.Text });
        }

        return Ok(chunks);
    }

    [HttpGet("chat/stream")]
    public IResult ChatStream(string prompt, CancellationToken cancellationToken)
    {
        return TypedResults.ServerSentEvents(StreamAnswer(prompt, cancellationToken));
    }

    private async IAsyncEnumerable<string> StreamAnswer(string prompt,
        [EnumeratorCancellation] CancellationToken cancellationToken)
    {
        List<ChatMessage> messages = [SystemPrompt(), new(ChatRole.User, prompt)];

        await foreach (ChatResponseUpdate update in
            _chat.GetStreamingResponseAsync(messages, ChatToolOptions(), cancellationToken))
        {
            if (!string.IsNullOrEmpty(update.Text))
                yield return update.Text;
        }
    }

    // not stored in the database: it is ours and is built anew every time (with today's date)
    private static ChatMessage SystemPrompt() => new(ChatRole.System, $"""
        You are a restaurant assistant. Answer briefly.
        Take menu and free-table data only from the tools, don't make anything up.
        Today is {DateTime.Now:yyyy-MM-dd, dddd}.
        """);

    private ChatOptions ChatToolOptions() => new()
    {
        Tools =
        [
            AIFunctionFactory.Create(_tools.GetMenu, name: "get_menu"),
            AIFunctionFactory.Create(_tools.CheckAvailability, name: "check_availability")
        ]
    };
}
```

```csharp Ai/LastTurnsReducer.cs — ⭐
public class LastTurnsReducer(int count) : IChatReducer
{
    public Task<IEnumerable<ChatMessage>> ReduceAsync(
        IEnumerable<ChatMessage> messages, CancellationToken cancellationToken)
    {
        IEnumerable<ChatMessage> textOnly = messages.Where(m =>
            !m.Contents.Any(c => c is FunctionCallContent or FunctionResultContent));
        return Task.FromResult(textOnly.TakeLast(count));
    }
}
```
