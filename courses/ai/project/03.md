- `POST /api/feedback/ai/sentiment` — `"Everything was perfect, we'll definitely come back!"` → `Positive` · `"The waiter was rude and the mtsvadi was burnt."` → `Negative` (კონსოლში: `SMS: notify the manager`) · `"Great wine, but the music was too loud."` → `Mixed`. პასუხი ყოველთვის ზუსტად ერთი მნიშვნელობაა.

```csharp Controllers/AiDiagnosticsController.cs
using RestaurantApi.Contracts.Diagnostics;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.AI;

namespace RestaurantApi.Controllers;

[ApiController, Route("api/diagnostics/ai")]
public sealed class AiDiagnosticsController(IChatClient chat) : ControllerBase
{
    [HttpPost("ask")]
    public async Task<IActionResult> Ask(AskRequest req)
    {
        var options = new ChatOptions { MaxOutputTokens = req.MaxOutputTokens, Temperature = req.Temperature };

        var response = await chat.GetResponseAsync(req.Message, options);

        return Ok(AskResponse.From(response));
    }

    [HttpPost("chat")]
    public async Task<IActionResult> Chat(ChatRequest req)
    {
        var messages = new List<ChatMessage>
        {
            new(ChatRole.System, SystemPrompt),
            new(ChatRole.User, req.Message)
        };

        var response = await chat.GetResponseAsync(messages);

        return Ok(new ChatReply(response.Text));
    }

    private const string SystemPrompt = """
                                        You are the assistant of the restaurant "Supra".
                                        You answer only about the menu, bookings and opening hours (12:00–23:00).
                                        Answer in at most 3 sentences, in the language the guest writes in.
                                        On any other topic, politely decline and say what you can help with.
                                        Never invent prices or ingredients — say they are on the menu.
                                        """;
}
```

```csharp Controllers/FeedbackController.cs
using RestaurantApi.Contracts.Feedback;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.AI;
using RestaurantApi.Enums;

namespace RestaurantApi.Controllers;

[ApiController, Route("api/feedback/ai")]
public sealed class FeedbackController(IChatClient chat) : ControllerBase
{
    [HttpPost("sentiment")]
    public async Task<IActionResult> AnalyzeSentiment(SentimentRequest req)
    {
        var messages = new List<ChatMessage>
        {
            new(ChatRole.System, "Classify the guest's review of the restaurant"),
            new(ChatRole.User, req.Message)
        };

        var response = await chat.GetResponseAsync<Sentiment>(messages);

        if (response.Result == Sentiment.Negative) Console.WriteLine("SMS: notify the manager");

        return Ok(new SentimentResponse(response.Result.ToString()));
    }
}
```

```csharp +Controllers/BookingsController.cs
using RestaurantApi.Contracts.Bookings;
using Microsoft.EntityFrameworkCore;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.AI;
using RestaurantApi.Entities;
using RestaurantApi.Data;

namespace RestaurantApi.Controllers;

[ApiController, Route("api/bookings")]
public sealed class BookingsController(DataContext db, IChatClient chat) : ControllerBase
{
    [HttpGet]
    public async Task<ActionResult<List<Booking>>> GetAll()
    {
        var bookings = await db.Bookings.OrderBy(b => b.At).ToListAsync();

        return bookings;
    }


    [HttpPost("parse")]
    public async Task<IActionResult> Parse(BookingParseRequest req)
    {
        var messages = new List<ChatMessage>
        {
            new(ChatRole.System, $"""
                                  Extract the booking details from the guest's message.
                                  Today is {DateTime.Now:yyyy-MM-dd, dddd}.
                                  Leave null whatever is not in the message.
                                  """),
            new(ChatRole.User, req.Message)
        };

        var response = await chat.GetResponseAsync<BookingParseResponse>(messages);

        if (response.Result.At is null)
            return BadRequest("Which day and what time?");

        if (response.Result.At.Value.Hour is < 12 or >= 23)
            return BadRequest("We're open 12:00–23:00");

        if (response.Result.Guests is null)
            return BadRequest("How many of you?");

        return Ok(response.Result);
    }
}
```
