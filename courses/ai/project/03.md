using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.AI;

[ApiController]
[Route("api/diagnostics/ai")]
public class AiDiagnosticsController : ControllerBase
{
    private readonly IChatClient _chat;

    public AiDiagnosticsController(IChatClient chat)
    {
        _chat = chat;
    }

    [HttpPost("ask")]
    public async Task<IActionResult> Ask(PromptRequest request)
    {
        var options = new ChatOptions
        {
            Temperature = request.Temperature,
            MaxOutputTokens = request.MaxOutputTokens
        };

        ChatResponse response = await _chat.GetResponseAsync(request.Prompt, options);
        return Ok(PromptResponse.From(response));
    }
}

```csharp Controllers/AssistantController.cs
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.AI;

[ApiController]
[Route("api/assistant")]
public class AssistantController : ControllerBase
{
    private readonly IChatClient _chat;

    public AssistantController(IChatClient chat)
    {
        _chat = chat;
    }

    [HttpPost("chat")]
    public async Task<IActionResult> Chat(ChatRequest request)
    {
        List<ChatMessage> messages =
        [
            SystemPrompt(),
            new(ChatRole.User, request.Message)
        ];

        ChatResponse response = await _chat.GetResponseAsync(messages);
        return Ok(new ChatReply(response.Text));
    }

    private static ChatMessage SystemPrompt() => new(ChatRole.System, """
        You are the assistant of the restaurant "Supra".
        You answer only about the menu, bookings and opening hours (12:00–23:00).
        Answer in at most 3 sentences, in the language the guest writes in.
        On any other topic, politely decline and say what you can help with.
        Never invent prices or ingredients — say they are on the menu.
        """);
}
```

```csharp Controllers/FeedbackController.cs
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.AI;

[ApiController]
[Route("api/feedback")]
public class FeedbackController : ControllerBase
{
    private readonly IChatClient _chat;

    public FeedbackController(IChatClient chat)
    {
        _chat = chat;
    }

    [HttpPost("sentiment")]
    public async Task<IActionResult> AnalyzeSentiment(SentimentRequest request)
    {
        List<ChatMessage> messages =
        [
            new(ChatRole.System, "Classify the guest's review of the restaurant."),
            new(ChatRole.User, request.Text)
        ];

        ChatResponse<Sentiment> response = await _chat.GetResponseAsync<Sentiment>(messages);

        Sentiment sentiment = response.Result;
        if (sentiment == Sentiment.Negative) { /* notify the manager */ }

        return Ok(new SentimentResponse(sentiment.ToString()));
    }
}
```

## ⭐ თუ დრო დარჩა: ყალბი `IChatClient`

DI-ის სარგებლის საჩვენებლად: `IChatClient`-ის ყალბი იმპლემენტაცია, რომელიც Gemini-ს არ იძახებს. `Program.cs`-ში დროებით `AddChatClient(new FakeChatClient("..."))`, და controller-ებში არაფერი იცვლება. პროექტში **არ** რჩება.

```csharp FakeChatClient.cs — მხოლოდ საჩვენებლად
public class FakeChatClient(string json) : IChatClient
{
    public Task<ChatResponse> GetResponseAsync(IEnumerable<ChatMessage> messages,
        ChatOptions? options = null, CancellationToken ct = default)
        => Task.FromResult(new ChatResponse(new ChatMessage(ChatRole.Assistant, json)));

    public IAsyncEnumerable<ChatResponseUpdate> GetStreamingResponseAsync(IEnumerable<ChatMessage> messages,
        ChatOptions? options = null, CancellationToken ct = default) => throw new NotImplementedException();

    public object? GetService(Type serviceType, object? serviceKey = null) => null;
    public void Dispose() { }
}
```
