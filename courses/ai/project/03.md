### ლექციაზე
- **ერთად გადააკეთებთ:** `Client` controller-იდან `Program.cs`-ში (`AddChatClient`), მოდელის სახელი — `appsettings.json`-ში, controller — constructor injection-ით.
- **ერთად წერთ:** `ReviewSentiment` → `GetResponseAsync<Sentiment>` (few-shot აღარ სჭირდება), შემდეგ `ParseBooking`.
- **წინასწარ:** ნაწილი 1 (DI) შენთან უკვე გაკეთებულია და `/ask` მუშაობს.

```note
`Program.cs`-ში, `appsettings.json`-ში და `secrets.json`-ში მხოლოდ AI-ის ნაწილი ჩანს — დანარჩენი შენი ფაილებია API კურსიდან (`AddDbContext`, `AddControllers`, Swagger…). `ApiKey` ისევ User Secrets-შია, `appsettings.json`-ში მხოლოდ მოდელის სახელია.
```

```csharp Controllers/AiLabController.cs
using System.ComponentModel;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.AI;

[ApiController]
[Route("api/ai-lab")]
public class AiLabController : ControllerBase
{
    private const string RestaurantName = "Supra";
    private const string OpenHours = "12:00–23:00";

    private readonly IChatClient _chat;

    public AiLabController(IChatClient chat)
    {
        _chat = chat;
    }

    [HttpPost("ask")]
    public async Task<IActionResult> Ask(AskRequest request)
    {
        var options = new ChatOptions
        {
            Temperature = request.Temperature,
            MaxOutputTokens = request.MaxOutputTokens
        };

        ChatResponse response =
            await _chat.GetResponseAsync(request.Prompt, options);
        return Ok(AskResponse.From(response));
    }

    [HttpPost("assistant")]
    public async Task<IActionResult> Assistant(AskRequest request)
    {
        List<ChatMessage> messages =
        [
            new(ChatRole.System, $"""
                You are the assistant of the restaurant "{RestaurantName}".
                You answer only about the menu, bookings and hours ({OpenHours}).
                Answer: at most 3 sentences, in the language the user writes in.
                On any other topic you politely decline and say what you can help with.
                You don't invent prices or ingredients — you say they are on the menu.
                """),
            new(ChatRole.User, request.Prompt)
        ];

        ChatResponse response = await _chat.GetResponseAsync(messages);
        return Ok(AskResponse.From(response));
    }

    public enum Sentiment { Positive, Negative, Mixed }

    [HttpPost("review-sentiment")]
    public async Task<IActionResult> ReviewSentiment(AskRequest request)
    {
        List<ChatMessage> messages =
        [
            new(ChatRole.System, "Classify the restaurant review."),
            new(ChatRole.User, request.Prompt)
        ];

        ChatResponse<Sentiment> response =
            await _chat.GetResponseAsync<Sentiment>(messages);

        Sentiment sentiment = response.Result;
        if (sentiment == Sentiment.Negative) { /* notify the manager */ }

        return Ok(sentiment.ToString());
    }

    public record BookingRequest(
        [property: Description("Booking date and time")] DateTime? At,
        [property: Description("Number of guests")] int? Guests);

    [HttpPost("parse-booking")]
    public async Task<IActionResult> ParseBooking(AskRequest request)
    {
        List<ChatMessage> messages =
        [
            new(ChatRole.System, $"""
                Extract the booking details from the user's text.
                Today is {DateTime.Now:yyyy-MM-dd, dddd}.
                Leave null whatever is not in the text.
                """),
            new(ChatRole.User, request.Prompt)
        ];

        ChatResponse<BookingRequest> response =
            await _chat.GetResponseAsync<BookingRequest>(messages);
        return Ok(response.Result);
    }
}
```

## ⭐ თუ დრო დარჩა: ყალბი `IChatClient`

DI-ის სარგებლის საჩვენებლად — `IChatClient`-ის ყალბი იმპლემენტაცია, რომელიც Gemini-ს არ იძახებს. `Program.cs`-ში დროებით `AddChatClient(new FakeChatClient("..."))` — და controller-ში არაფერი იცვლება. პროექტში **არ** რჩება.

```csharp FakeChatClient.cs — მხოლოდ საჩვენებლად
public class FakeChatClient(string json) : IChatClient
{
    public Task<ChatResponse> GetResponseAsync(IEnumerable<ChatMessage> messages,
        ChatOptions? options = null, CancellationToken ct = default)
        => Task.FromResult(new ChatResponse(new ChatMessage(ChatRole.Assistant, json)));

    public IAsyncEnumerable<ChatResponseUpdate> GetStreamingResponseAsync(IEnumerable<ChatMessage> messages,
        ChatOptions? options = null, CancellationToken ct = default) => throw new NotImplementedException();

    public object? GetService(Type serviceType, object? serviceKey = null) => null;
    public void Dispose() { }
}
```
