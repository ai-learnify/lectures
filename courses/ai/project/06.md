- **ერთად წერთ:** `AiDiagnosticsController.StreamTiming`: `await foreach` და `Stopwatch`, როდის მოდის თითო ნაწილი.
- **ჩასვამ და ახსნი:** `AssistantController.ChatStream` (`TypedResults.ServerSentEvents` + `StreamAnswer`).
- **წინასწარ:** ორივე შენთან დაწერილი და გაშვებულია; curl-ის ბრძანებები ფაილში (მოკლე, tool-იანი, გრძელი შეკითხვა).

### შემოწმება (curl)
Swagger სტრიმს ვერ აჩვენებს: მთლიან პასუხს ბოლოს გამოიტანს. ამიტომ `curl -N` (`-N` — buffering-ის გარეშე):
```bash
curl -N "https://localhost:7001/api/assistant/chat/stream?message=Tell%20us%20about%20Georgian%20cuisine%20in%20detail"
curl -N "https://localhost:7001/api/assistant/chat/stream?message=What%20do%20you%20have%20under%2015%20GEL%3F"
```
პორტი შენი პროექტის `launchSettings.json`-იდანაა. სტრიმის შუაში **Ctrl+C** → ლოგში Gemini-ს მოთხოვნა `canceled`-ია: `CancellationToken` სამ ადგილას მუშაობს, endpoint → `StreamAnswer` (`[EnumeratorCancellation]`) → `GetStreamingResponseAsync`.

```csharp Controllers/AssistantController.cs
using System.Runtime.CompilerServices;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.AI;

[ApiController]
[Route("api/assistant")]
public class AssistantController : ControllerBase
{
    private readonly IChatClient _chat;
    private readonly RestaurantTools _tools;
    private readonly ILogger<AssistantController> _logger;

    public AssistantController(IChatClient chat, RestaurantTools tools, ILogger<AssistantController> logger)
    {
        _chat = chat;
        _tools = tools;
        _logger = logger;
    }

    [HttpPost("chat")]
    public async Task<IActionResult> Chat(ChatRequest request)
    {
        List<ChatMessage> messages =
        [
            SystemPrompt(),
            new(ChatRole.User, request.Message)
        ];

        ChatResponse response = await _chat.GetResponseAsync(messages, CreateChatOptions());

        List<string> toolCalls = response.Messages
            .SelectMany(m => m.Contents)
            .OfType<FunctionCallContent>()
            .Select(c => c.Name)
            .ToList();
        _logger.LogInformation("Tools called: {Tools}", string.Join(", ", toolCalls));

        return Ok(new ChatReply(response.Text));
    }

    [HttpGet("chat/stream")]
    public IResult ChatStream(string message, CancellationToken cancellationToken)
    {
        return TypedResults.ServerSentEvents(StreamAnswer(message, cancellationToken));
    }

    private async IAsyncEnumerable<string> StreamAnswer(string message,
        [EnumeratorCancellation] CancellationToken cancellationToken)
    {
        List<ChatMessage> messages = [SystemPrompt(), new(ChatRole.User, message)];

        await foreach (ChatResponseUpdate update in
            _chat.GetStreamingResponseAsync(messages, CreateChatOptions(), cancellationToken))
        {
            if (!string.IsNullOrEmpty(update.Text))
                yield return update.Text;
        }
    }

    private ChatOptions CreateChatOptions() => new()
    {
        Tools =
        [
            AIFunctionFactory.Create(_tools.GetMenu, name: "get_menu"),
            AIFunctionFactory.Create(_tools.CheckAvailability, name: "check_availability")
        ]
    };

    private static ChatMessage SystemPrompt() => new(ChatRole.System, $"""
        You are the assistant of the restaurant "Supra".
        You answer only about the menu, bookings and opening hours (12:00–23:00).
        Answer in at most 3 sentences, in the language the guest writes in.
        Take menu and free-table data only from the tools, never make anything up.
        Today is {DateTime.Now:yyyy-MM-dd, dddd}.
        """);
}
```

```csharp Controllers/AiDiagnosticsController.cs
using System.Diagnostics;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.AI;

[ApiController]
[Route("api/diagnostics/ai")]
public class AiDiagnosticsController : ControllerBase
{
    private readonly IChatClient _chat;

    public AiDiagnosticsController(IChatClient chat)
    {
        _chat = chat;
    }

    [HttpPost("ask")]
    public async Task<IActionResult> Ask(PromptRequest request)
    {
        var options = new ChatOptions
        {
            Temperature = request.Temperature,
            MaxOutputTokens = request.MaxOutputTokens
        };

        ChatResponse response = await _chat.GetResponseAsync(request.Prompt, options);
        return Ok(PromptResponse.From(response));
    }

    [HttpPost("stream-timing")]
    public async Task<IActionResult> StreamTiming(PromptRequest request)
    {
        Stopwatch sw = Stopwatch.StartNew();
        List<object> chunks = [];

        await foreach (ChatResponseUpdate update in _chat.GetStreamingResponseAsync(request.Prompt))
        {
            chunks.Add(new { Ms = sw.ElapsedMilliseconds, update.Text });
        }

        return Ok(chunks);
    }
}
```

## ⭐ თუ დრო დარჩა: ბრაუზერში — `EventSource`

    out.textContent = '';
    const es = new EventSource('/api/assistant/chat/stream?message=' + encodeURIComponent(q.value));
