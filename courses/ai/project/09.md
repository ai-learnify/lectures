### შემოწმება (Swagger)
1. `POST /api/diagnostics/ai/embed` `{ "text": "lobio" }` → `length: 768` და პირველი 5 რიცხვი.
2. `POST /api/diagnostics/ai/similarity` `{ "first": "lobio", "second": "bean stew" }` → score ~0.8; `"lobio"` და `"car tires"` → ბევრად ნაკლები.
3. `POST /api/menu/embeddings` → `{ "count": N }`.
4. `GET /api/menu/search?query=something light without meat` → top 3 კერძი score-ით (მაგ. ლობიო): ძებნა მნიშვნელობით, არა სიტყვით.

```csharp Controllers/AiDiagnosticsController.cs
using System.Diagnostics;
using System.Numerics.Tensors;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.AI;

[ApiController]
[Route("api/diagnostics/ai")]
[DevelopmentOnly]
public class AiDiagnosticsController : ControllerBase
{
    private readonly IChatClient _chat;
    private readonly IEmbeddingGenerator<string, Embedding<float>> _embeddings;

    public AiDiagnosticsController(IChatClient chat,
        IEmbeddingGenerator<string, Embedding<float>> embeddings)
    {
        _chat = chat;
        _embeddings = embeddings;
    }

    [HttpPost("ask")]
    public async Task<IActionResult> Ask(PromptRequest request)
    {
        var options = new ChatOptions
        {
            Temperature = request.Temperature,
            MaxOutputTokens = request.MaxOutputTokens
        };

        ChatResponse response = await _chat.GetResponseAsync(request.Prompt, options);
        return Ok(PromptResponse.From(response));
    }

    [HttpPost("stream-timing")]
    public async Task<IActionResult> StreamTiming(PromptRequest request)
    {
        Stopwatch sw = Stopwatch.StartNew();
        List<object> chunks = [];

        await foreach (ChatResponseUpdate update in _chat.GetStreamingResponseAsync(request.Prompt))
        {
            chunks.Add(new { Ms = sw.ElapsedMilliseconds, update.Text });
        }

        return Ok(chunks);
    }

    [HttpPost("embed")]
    public async Task<IActionResult> Embed(EmbedRequest request, CancellationToken ct)
    {
        ReadOnlyMemory<float> vector =
            await _embeddings.GenerateVectorAsync(request.Text, cancellationToken: ct);

        return Ok(new EmbedResponse(vector.Length, vector.ToArray()[..5]));
    }

    [HttpPost("similarity")]
    public async Task<IActionResult> Similarity(SimilarityRequest request, CancellationToken ct)
    {
        GeneratedEmbeddings<Embedding<float>> vectors = await _embeddings.GenerateAsync(
            [request.First, request.Second], cancellationToken: ct);

        float score = TensorPrimitives.CosineSimilarity(
            vectors[0].Vector.Span, vectors[1].Vector.Span);

        return Ok(new SimilarityResponse(score));
    }
}
```

```csharp +Controllers/MenuController.cs
using System.Numerics.Tensors;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.AI;

[ApiController]
[Route("api/menu")]
public class MenuController : ControllerBase
{
    // temporary: the vectors in memory, lost when the app restarts (lecture 10 → database)
    private static List<(string Name, ReadOnlyMemory<float> Vector)> _vectors = [];

    private readonly DataContext _db;
    private readonly IEmbeddingGenerator<string, Embedding<float>> _embeddings;

    public MenuController(DataContext db, IEmbeddingGenerator<string, Embedding<float>> embeddings)
    {
        _db = db;
        _embeddings = embeddings;
    }

    // ... your endpoints from the API course

    // ---------- AI ----------
    [HttpPost("embeddings")]
    public async Task<IActionResult> GenerateEmbeddings(CancellationToken ct)
    {
        List<Product> products = await _db.Products.ToListAsync(ct);

        GeneratedEmbeddings<Embedding<float>> vectors = await _embeddings.GenerateAsync(
            products.Select(p => EmbeddingInput.Document(p.Name, p.Description)),
            cancellationToken: ct);

        _vectors = products
            .Select((p, i) => (p.Name, vectors[i].Vector))
            .ToList();

        return Ok(new EmbeddingsResult(products.Count));
    }

    [HttpGet("search")]
    public async Task<IActionResult> Search(string query, CancellationToken ct)
    {
        ReadOnlyMemory<float> queryVector = await _embeddings.GenerateVectorAsync(
            EmbeddingInput.Query(query), cancellationToken: ct);

        List<MenuMatch> top = _vectors
            .Select(v => new MenuMatch(v.Name,
                TensorPrimitives.CosineSimilarity(queryVector.Span, v.Vector.Span)))
            .OrderByDescending(m => m.Score)
            .Take(3)
            .ToList();

        return Ok(top);
    }
}
```
