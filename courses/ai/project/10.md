```csharp +Controllers/MenuController.cs
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.AI;
using Pgvector;
using Pgvector.EntityFrameworkCore;

[ApiController]
[Route("api/menu")]
public class MenuController : ControllerBase
{
    private readonly DataContext _db;
    private readonly IEmbeddingGenerator<string, Embedding<float>> _embeddings;
    private readonly IConfiguration _config;

    public MenuController(DataContext db, IEmbeddingGenerator<string, Embedding<float>> embeddings,
        IConfiguration config)
    {
        _db = db;
        _embeddings = embeddings;
        _config = config;
    }

    // ... your endpoints from the API course

    // ---------- AI ----------
    [HttpPost("embeddings")]
    public async Task<IActionResult> GenerateEmbeddings(CancellationToken ct)
    {
        string model = _config["Gemini:EmbeddingModel"]!;

        // only the dishes that have no vector, or have one from another model
        List<Product> products = await _db.Products
            .Where(p => p.Embedding == null || p.EmbeddingModel != model)
            .ToListAsync(ct);

        foreach (Product[] batch in products.Chunk(100))          // Gemini: up to 100 per request
        {
            GeneratedEmbeddings<Embedding<float>> vectors = await _embeddings.GenerateAsync(
                batch.Select(p => EmbeddingInput.Document(p.Name, p.Description)),
                cancellationToken: ct);

            for (int i = 0; i < batch.Length; i++)
            {
                batch[i].Embedding = new Vector(vectors[i].Vector);  // ReadOnlyMemory<float> → Vector
                batch[i].EmbeddingModel = model;
            }
        }

        await _db.SaveChangesAsync(ct);
        return Ok(new EmbeddingsResult(products.Count));
    }

    [HttpGet("search")]
    public async Task<IActionResult> Search(string query, CancellationToken ct)
    {
        var queryVector = new Vector(await _embeddings.GenerateVectorAsync(
            EmbeddingInput.Query(query), cancellationToken: ct));

        List<MenuMatch> top = await _db.Products
            .Where(p => p.Embedding != null)
            .OrderBy(p => p.Embedding!.CosineDistance(queryVector))  // → ORDER BY "Embedding" <=> @query
            .Take(3)                                                   // → LIMIT 3
            .Select(p => new MenuMatch(p.Name, 1 - p.Embedding!.CosineDistance(queryVector)))
            .ToListAsync(ct);

        return Ok(top);
    }
}
```

```csharp Ai/Tools/RestaurantTools.cs · SearchMenu — ⭐
[Description("Finds up to 3 menu dishes that match a description, e.g. 'something light without meat'.")]
public async Task<string> SearchMenu(
    [Description("What the guest wants, in their own words")] string query)
{
    var queryVector = new Vector(await _embeddings.GenerateVectorAsync(EmbeddingInput.Query(query)));

    List<string> names = await _db.Products
        .Where(p => p.Embedding != null)
        .OrderBy(p => p.Embedding!.CosineDistance(queryVector))
        .Take(3)
        .Select(p => p.Name)
        .ToListAsync();

    return string.Join(", ", names);
}
```
