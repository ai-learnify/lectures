```csharp Ai/Documents/DocumentSearch.cs
using System.Text.RegularExpressions;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.AI;
using Pgvector;
using Pgvector.EntityFrameworkCore;

public class DocumentSearch
{
    private readonly DataContext _db;
    private readonly IEmbeddingGenerator<string, Embedding<float>> _embeddings;
    private readonly float _minScore;

    public DocumentSearch(DataContext db, IEmbeddingGenerator<string, Embedding<float>> embeddings,
        IConfiguration config)
    {
        _db = db;
        _embeddings = embeddings;
        _minScore = config.GetValue<float>("Rag:MinScore");   // chosen by testing, like lecture 9's homework
    }

    // hybrid: by meaning + by words, merged with RRF
    public async Task<List<ChunkMatch>> SearchAsync(string query, string? category, CancellationToken ct)
    {
        var queryVector = new Vector(await _embeddings.GenerateVectorAsync(
            EmbeddingInput.Query(query), cancellationToken: ct));
        var words = EF.Functions.ToTsQuery("english", OrQuery(query));

        // metadata filtering — always in SQL, before the search
        IQueryable<Chunk> chunks = _db.Chunks;
        if (category is not null)
            chunks = chunks.Where(c => c.Document.Category == category);

        List<int> byMeaning = await chunks                                 // by vector, with a threshold
            .Where(c => 1 - c.Embedding!.CosineDistance(queryVector) >= _minScore)
            .OrderBy(c => c.Embedding!.CosineDistance(queryVector))
            .Select(c => c.Id)
            .Take(20)
            .ToListAsync(ct);

        List<int> byWords = await chunks                                   // by words
            .Where(c => c.SearchVector.Matches(words))
            .OrderByDescending(c => c.SearchVector.Rank(words))
            .Select(c => c.Id)
            .Take(20)
            .ToListAsync(ct);

        List<int> top = Rrf(byMeaning, byWords).Take(5).ToList();

        var found = await _db.Chunks
            .Where(c => top.Contains(c.Id))
            .Select(c => new { c.Id, c.Document.FileName, c.Page, c.Content })
            .ToDictionaryAsync(c => c.Id, ct);

        return top
            .Select((id, i) => new ChunkMatch(found[id].FileName, found[id].Page, found[id].Content,
                Score: 1.0 / (i + 1)))                                     // Score — here simply by rank
            .ToList();
    }

    // lecture 11: by vector only — kept to compare in the evaluation (?hybrid=false)
    public async Task<List<ChunkMatch>> VectorSearchAsync(string query, CancellationToken ct)
    {
        var queryVector = new Vector(await _embeddings.GenerateVectorAsync(
            EmbeddingInput.Query(query), cancellationToken: ct));

        List<ChunkMatch> top = await _db.Chunks
            .Where(c => c.Embedding != null)
            .OrderBy(c => c.Embedding!.CosineDistance(queryVector))
            .Take(5)
            .Select(c => new ChunkMatch(c.Document.FileName, c.Page, c.Content,
                1 - c.Embedding!.CosineDistance(queryVector)))
            .ToListAsync(ct);

        return top.Where(c => c.Score >= _minScore).ToList();
    }

    // the found parts as text for the model: "[1] (rules.pdf, p. 3)" + the text
    public static string FormatContext(List<ChunkMatch> chunks) =>
        chunks.Count == 0
            ? "(No relevant information was found in the sources)"
            : string.Join("\n\n", chunks.Select((c, i) => $"[{i + 1}] ({c.File}, p. {c.Page})\n{c.Content}"));

    private static string OrQuery(string query)
    {
        var words = Regex.Matches(query, @"[\p{L}\p{N}]+(?:-[\p{L}\p{N}]+)*").Select(m => m.Value);
        return string.Join(" | ", words);   // "Where is hall B-2?" → "Where | is | hall | B-2"
    }

    private static IEnumerable<int> Rrf(params List<int>[] lists) => lists
        .SelectMany(list => list.Select((id, i) => (id, score: 1.0 / (60 + i + 1))))
        .GroupBy(x => x.id)
        .OrderByDescending(g => g.Sum(x => x.score))
        .Select(g => g.Key);
}
```

```csharp Controllers/DocumentsController.cs
    private readonly DataContext _db;
    private readonly IEmbeddingGenerator<string, Embedding<float>> _embeddings;
    private readonly DocumentSearch _search;
    private readonly IConfiguration _config;

    public DocumentsController(DataContext db, IEmbeddingGenerator<string, Embedding<float>> embeddings,
        DocumentSearch search, IConfiguration config)
    {
        _db = db;
        _embeddings = embeddings;
        _search = search;
        _config = config;
    }

    [HttpPost]
    public async Task<IActionResult> Upload(IFormFile file, [FromForm] string category = "rules",
        CancellationToken ct = default)
    {
        using var ms = new MemoryStream();
        await file.CopyToAsync(ms, ct);

        // 1–2: PDF → pages → parts
        var document = new Document
        {
            FileName = file.FileName,
            Category = category,
            UploadedAt = DateTime.UtcNow
        };
        foreach (var (page, text) in PdfTextReader.ReadPages(ms.ToArray()))
            foreach (string content in TextChunker.Split(text))
                document.Chunks.Add(new Chunk { Page = page, Content = content });

        // 3: vectors, in batches of 100
        string model = _config["Gemini:EmbeddingModel"]!;
        foreach (Chunk[] batch in document.Chunks.Chunk(100))
        {
            GeneratedEmbeddings<Embedding<float>> vectors = await _embeddings.GenerateAsync(
                batch.Select(c => EmbeddingInput.Document(document.FileName, c.Content)),
                cancellationToken: ct);

            for (int i = 0; i < batch.Length; i++)
            {
                batch[i].Embedding = new Vector(vectors[i].Vector);
                batch[i].EmbeddingModel = model;
            }
        }

        // 4: into the database — the document and all its parts together
        _db.Documents.Add(document);
        await _db.SaveChangesAsync(ct);

        return Ok(new DocumentUploaded(document.Id, document.Chunks.Count));
    }

    [HttpGet("search")]
    public async Task<IActionResult> Search(string query, string? category, CancellationToken ct)
    {
        return Ok(await _search.SearchAsync(query, category, ct));
    }
}
```

```csharp Controllers/EvaluationController.cs
using System.Text.Json;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.AI;
using Microsoft.Extensions.AI.Evaluation;
using Microsoft.Extensions.AI.Evaluation.Quality;

[ApiController]
[Route("api/evaluation")]
public class EvaluationController : ControllerBase
{
    private readonly DocumentSearch _search;
    private readonly IChatClient _chat;
    private readonly IChatClient _judge;

    public EvaluationController(DocumentSearch search, IChatClient chat,
        [FromKeyedServices("judge")] IChatClient judge)
    {
        _search = search;
        _chat = chat;
        _judge = judge;
    }

    [HttpPost("retrieval")]
    public async Task<IActionResult> Retrieval(bool hybrid = true, CancellationToken ct = default)
    {
        string json = await System.IO.File.ReadAllTextAsync("golden.json", ct);   // ControllerBase.File is something else
        List<GoldenItem> golden = JsonSerializer.Deserialize<List<GoldenItem>>(json, JsonSerializerOptions.Web)!;

        int hits = 0;
        double reciprocalRanks = 0;
        List<string> misses = [];

        foreach (GoldenItem item in golden)
        {
            List<ChunkMatch> found = hybrid
                ? await _search.SearchAsync(item.Question, null, ct)
                : await _search.VectorSearchAsync(item.Question, ct);

            int rank = found.FindIndex(c => c.Content.Contains(item.MustContain)) + 1;   // 1, 2, 3… or 0

            if (rank is >= 1 and <= 3) hits++;
            else misses.Add(item.Question);

            if (rank > 0) reciprocalRanks += 1.0 / rank;
        }

        return Ok(new RetrievalReport(
            HitAt3: (double)hits / golden.Count,
            Mrr: reciprocalRanks / golden.Count,
            Misses: misses));
    }

    [HttpPost("groundedness")]
    public async Task<IActionResult> Groundedness(GroundednessRequest request, CancellationToken ct)
    {
        List<ChunkMatch> sources = await _search.SearchAsync(request.Question, null, ct);
        string context = DocumentSearch.FormatContext(sources);

        List<ChatMessage> messages =
        [
            new(ChatRole.System, """
                Answer only from the sources and give the number: [1].
                If it isn't in the sources, say: "I don't have information on this".
                """),
            new(ChatRole.User, $"""
                Sources:
                {context}

                Question: {request.Question}
                """)
        ];

        // without tools: the answer must come only from these sources
        ChatResponse response = await _chat.GetResponseAsync(messages, cancellationToken: ct);

        IEvaluator evaluator = new GroundednessEvaluator();
        EvaluationResult result = await evaluator.EvaluateAsync(
            [new ChatMessage(ChatRole.User, request.Question)],      // what was asked
            response,                                                // what our RAG answered
            new ChatConfiguration(_judge),                           // the judge — an IChatClient
            [new GroundednessEvaluatorContext(context)],             // the sources the model received
            ct);

        NumericMetric score = result.Get<NumericMetric>(GroundednessEvaluator.GroundednessMetricName);

        return Ok(new GroundednessReport(response.Text, score.Value, score.Reason));   // 1–5 and why
    }
}
```

```csharp Ai/Documents/DocumentSearch.cs · RerankAsync — ⭐
public record RankResult(List<int> Best);

private async Task<List<ChunkMatch>> RerankAsync(string query, List<ChunkMatch> candidates,
    CancellationToken ct)
{
    string list = string.Join("\n\n", candidates.Select((c, i) => $"[{i}] {c.Content}"));

    ChatResponse<RankResult> response = await _judge.GetResponseAsync<RankResult>($"""
        Question: {query}

        Below are the chunks. Return the numbers of the 5 chunks that answer
        this question best, in decreasing order of relevance.

        {list}
        """, cancellationToken: ct);

    return response.Result.Best
        .Where(i => i >= 0 && i < candidates.Count)
        .Distinct()
        .Take(5)
        .Select(i => candidates[i])
        .ToList();
}
```
