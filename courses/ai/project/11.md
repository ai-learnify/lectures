```csharp Ai/Documents/PdfTextReader.cs
using System.Text.RegularExpressions;
using UglyToad.PdfPig;
using UglyToad.PdfPig.DocumentLayoutAnalysis.TextExtractor;

public static class PdfTextReader
{
    public static List<(int Page, string Text)> ReadPages(byte[] pdf)
    {
        using PdfDocument doc = PdfDocument.Open(pdf);
        return doc.GetPages()
            .Select(p => (p.Number, Clean(ContentOrderTextExtractor.GetText(p))))
            .Where(p => p.Item2.Length > 0)          // skip empty pages
            .ToList();
    }

    private static string Clean(string text) =>
        Regex.Replace(text, @"\s+", " ").Trim();     // extra spaces and line breaks
}
```

```csharp Ai/Documents/TextChunker.cs
using System.Text.RegularExpressions;

public static class TextChunker
{
    public static List<string> Split(string text, int maxChars = 800, int overlap = 1)
    {
        string[] sentences = Regex.Split(text, @"(?<=[.!?])\s+");
        List<string> chunks = [];
        List<string> current = [];

        foreach (string sentence in sentences)
        {
            if (current.Count > 0 && current.Sum(s => s.Length) + sentence.Length > maxChars)
            {
                chunks.Add(string.Join(" ", current));
                current = current.TakeLast(overlap).ToList();   // overlap: the last sentence stays
            }
            current.Add(sentence);
        }
        if (current.Count > 0)
            chunks.Add(string.Join(" ", current));
        return chunks;
    }
}
```

```csharp Ai/Documents/DocumentSearch.cs
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.AI;
using Pgvector;
using Pgvector.EntityFrameworkCore;

public class DocumentSearch
{
    private readonly DataContext _db;
    private readonly IEmbeddingGenerator<string, Embedding<float>> _embeddings;
    private readonly float _minScore;

    public DocumentSearch(DataContext db, IEmbeddingGenerator<string, Embedding<float>> embeddings,
        IConfiguration config)
    {
        _db = db;
        _embeddings = embeddings;
        _minScore = config.GetValue<float>("Rag:MinScore");   // chosen by testing, like lecture 9's homework
    }

    public async Task<List<ChunkMatch>> SearchAsync(string query, CancellationToken ct)
    {
        var queryVector = new Vector(await _embeddings.GenerateVectorAsync(
            EmbeddingInput.Query(query), cancellationToken: ct));

        List<ChunkMatch> top = await _db.Chunks
            .Where(c => c.Embedding != null)
            .OrderBy(c => c.Embedding!.CosineDistance(queryVector))
            .Take(5)
            .Select(c => new ChunkMatch(c.Document.FileName, c.Page, c.Content,
                1 - c.Embedding!.CosineDistance(queryVector)))
            .ToListAsync(ct);

        return top.Where(c => c.Score >= _minScore).ToList();
    }

    // the found parts as text for the model: "[1] (rules.pdf, p. 3)" + the text
    public static string FormatContext(List<ChunkMatch> chunks) =>
        chunks.Count == 0
            ? "(No relevant information was found in the sources)"
            : string.Join("\n\n", chunks.Select((c, i) => $"[{i + 1}] ({c.File}, p. {c.Page})\n{c.Content}"));
}
```

```csharp Controllers/DocumentsController.cs
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.AI;
using Pgvector;

[ApiController]
[Route("api/documents")]
public class DocumentsController : ControllerBase
{
    private readonly DataContext _db;
    private readonly IEmbeddingGenerator<string, Embedding<float>> _embeddings;
    private readonly DocumentSearch _search;
    private readonly IConfiguration _config;

    public DocumentsController(DataContext db, IEmbeddingGenerator<string, Embedding<float>> embeddings,
        DocumentSearch search, IConfiguration config)
    {
        _db = db;
        _embeddings = embeddings;
        _search = search;
        _config = config;
    }

    [HttpPost]
    public async Task<IActionResult> Upload(IFormFile file, CancellationToken ct)
    {
        using var ms = new MemoryStream();
        await file.CopyToAsync(ms, ct);

        // 1–2: PDF → pages → parts
        var document = new Document { FileName = file.FileName, UploadedAt = DateTime.UtcNow };
        foreach (var (page, text) in PdfTextReader.ReadPages(ms.ToArray()))
            foreach (string content in TextChunker.Split(text))
                document.Chunks.Add(new Chunk { Page = page, Content = content });

        // 3: vectors, in batches of 100
        string model = _config["Gemini:EmbeddingModel"]!;
        foreach (Chunk[] batch in document.Chunks.Chunk(100))
        {
            GeneratedEmbeddings<Embedding<float>> vectors = await _embeddings.GenerateAsync(
                batch.Select(c => EmbeddingInput.Document(document.FileName, c.Content)),
                cancellationToken: ct);

            for (int i = 0; i < batch.Length; i++)
            {
                batch[i].Embedding = new Vector(vectors[i].Vector);
                batch[i].EmbeddingModel = model;
            }
        }

        // 4: into the database — the document and all its parts together
        _db.Documents.Add(document);
        await _db.SaveChangesAsync(ct);

        return Ok(new DocumentUploaded(document.Id, document.Chunks.Count));
    }

    [HttpGet("search")]
    public async Task<IActionResult> Search(string query, CancellationToken ct)
    {
        return Ok(await _search.SearchAsync(query, ct));
    }
}
```

```csharp Controllers/AssistantController.cs
using System.Runtime.CompilerServices;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.AI;

[ApiController]
[Route("api/assistant")]
public class AssistantController : ControllerBase
{
#pragma warning disable MEAI001   // reducers are still experimental
    private static readonly IChatReducer Reducer = new MessageCountingChatReducer(10);
#pragma warning restore MEAI001

    private readonly IChatClient _chat;
    private readonly RestaurantTools _tools;
    private readonly ChatHistoryStore _history;
    private readonly DocumentSearch _documents;
    private readonly ILogger<AssistantController> _logger;

    public AssistantController(IChatClient chat, RestaurantTools tools, ChatHistoryStore history,
        DocumentSearch documents, ILogger<AssistantController> logger)
    {
        _chat = chat;
        _tools = tools;
        _history = history;
        _documents = documents;
        _logger = logger;
    }

    [HttpPost("chat")]
    public async Task<IActionResult> Chat(ChatRequest request, CancellationToken ct)
    {
        Guid conversationId = request.ConversationId ?? Guid.NewGuid();

        List<ChatMessage> history = await _history.LoadAsync(conversationId, ct);
        IEnumerable<ChatMessage> recent = await Reducer.ReduceAsync(history, ct);

        List<ChunkMatch> sources = await _documents.SearchAsync(request.Message, ct);   // retrieve

        ChatMessage question = new(ChatRole.User, request.Message);
        ChatMessage augmented = new(ChatRole.User, $"""
            Sources:
            {DocumentSearch.FormatContext(sources)}

            Question: {request.Message}
            """);                                                                        // augment

        List<ChatMessage> messages = [SystemPrompt(), .. recent, augmented];

        ChatResponse response =
            await _chat.GetResponseAsync(messages, CreateChatOptions(), ct);              // generate

        // into the history — the original question, not the one with the sources
        await _history.AppendAsync(conversationId, [question, .. response.Messages], ct);

        List<string> toolCalls = response.Messages
            .SelectMany(m => m.Contents)
            .OfType<FunctionCallContent>()
            .Select(c => c.Name)
            .ToList();
        _logger.LogInformation("Tools called: {Tools}", string.Join(", ", toolCalls));

        List<ChatSource> sourceList = sources
            .Select((s, i) => new ChatSource(i + 1, s.File, s.Page))
            .ToList();

        return Ok(new ChatReply(conversationId, response.Text, sourceList));
    }

    [HttpGet("chat/stream")]
    public IResult ChatStream(string message, CancellationToken cancellationToken)
    {
        return TypedResults.ServerSentEvents(StreamAnswer(message, cancellationToken));
    }

    private async IAsyncEnumerable<string> StreamAnswer(string message,
        [EnumeratorCancellation] CancellationToken cancellationToken)
    {
        List<ChatMessage> messages = [SystemPrompt(), new(ChatRole.User, message)];

        await foreach (ChatResponseUpdate update in
            _chat.GetStreamingResponseAsync(messages, CreateChatOptions(), cancellationToken))
        {
            if (!string.IsNullOrEmpty(update.Text))
                yield return update.Text;
        }
    }

    private ChatOptions CreateChatOptions() => new()
    {
        Tools =
        [
            AIFunctionFactory.Create(_tools.GetMenu, name: "get_menu"),
            AIFunctionFactory.Create(_tools.CheckAvailability, name: "check_availability")
        ]
    };

    // not stored in the database: it is ours and is built anew every time (with today's date)
    private static ChatMessage SystemPrompt() => new(ChatRole.System, $"""
        You are the assistant of the restaurant "Supra".
        You answer only about the menu, bookings and opening hours (12:00–23:00).
        Answer in at most 3 sentences, in the language the guest writes in.
        Take menu and free-table data only from the tools, never make anything up.
        On the restaurant's rules answer only from the sources and give the number: [1].
        If it isn't in the sources, say: "I don't have information on this".
        Today is {DateTime.Now:yyyy-MM-dd, dddd}.
        """);
}
```
