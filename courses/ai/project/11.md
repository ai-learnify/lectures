### ლექციაზე
- **ერთად წერთ:** `UploadDocument` (ingestion: PDF → გვერდები → chunk-ები → ვექტორები → ბაზა) და `/chat`-ის ცვლილება (retrieve → augment → generate).
- **ჩასვამ და ახსნი:** `PdfText`, `TextChunker`, `Document`/`DocumentChunk` და `SearchChunksAsync`.
- **წინასწარ:** ყველაფერი შენთან მუშაობს; migration HNSW ინდექსით; `Rag:MinScore` შენს PDF-ზე შერჩეული; სატესტო PDF (რესტორნის წესები, ინგლისურად).

### შემოწმება (Swagger)
1. `/documents` — PDF-ის ატვირთვა → `{ id, chunks: N }`.
2. `/chat` `"How do I cancel a booking?"` → პასუხი `[1]`-ით და `sources: [{ n: 1, file: "rules.pdf", page: 3 }]`.
3. `/chat` `"Do you have parking?"` (PDF-ში არ წერია) → „I don't have information on this“, `sources` ცარიელი.
4. `/chat` `"How much is lobio?"` → პასუხი tool-იდან (მენიუ), არა PDF-იდან.

```csharp Ai/PdfText.cs
using System.Text.RegularExpressions;
using UglyToad.PdfPig;
using UglyToad.PdfPig.DocumentLayoutAnalysis.TextExtractor;

public static class PdfText
{
    public static List<(int Page, string Text)> ReadPages(byte[] pdf)
    {
        using PdfDocument doc = PdfDocument.Open(pdf);
        return doc.GetPages()
            .Select(p => (p.Number, Clean(ContentOrderTextExtractor.GetText(p))))
            .Where(p => p.Item2.Length > 0)          // skip empty pages
            .ToList();
    }

    private static string Clean(string text) =>
        Regex.Replace(text, @"\s+", " ").Trim();     // extra spaces and line breaks
}
```

```csharp Ai/TextChunker.cs
using System.Text.RegularExpressions;

public static class TextChunker
{
    public static List<string> Split(string text, int maxChars = 800, int overlap = 1)
    {
        string[] sentences = Regex.Split(text, @"(?<=[.!?])\s+");
        List<string> chunks = [];
        List<string> current = [];

        foreach (string sentence in sentences)
        {
            if (current.Count > 0 && current.Sum(s => s.Length) + sentence.Length > maxChars)
            {
                chunks.Add(string.Join(" ", current));
                current = current.TakeLast(overlap).ToList();   // overlap: the last sentence stays
            }
            current.Add(sentence);
        }
        if (current.Count > 0)
            chunks.Add(string.Join(" ", current));
        return chunks;
    }
}
```

```csharp Controllers/AiLabController.cs
using System.ComponentModel;
using System.Diagnostics;
using System.Runtime.CompilerServices;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.AI;
using Pgvector;
using Pgvector.EntityFrameworkCore;

[ApiController]
[Route("api/ai-lab")]
public class AiLabController : ControllerBase
{
    private const string RestaurantName = "Supra";
    private const string OpenHours = "12:00–23:00";

#pragma warning disable MEAI001   // reducers are still experimental
    private static readonly IChatReducer Reducer = new MessageCountingChatReducer(10);
#pragma warning restore MEAI001

    private readonly IChatClient _chat;
    private readonly RestaurantTools _tools;
    private readonly ChatHistoryStore _history;
    private readonly IEmbeddingGenerator<string, Embedding<float>> _embeddings;
    private readonly DataContext _db;
    private readonly IConfiguration _config;

    public AiLabController(IChatClient chat, RestaurantTools tools, ChatHistoryStore history,
        IEmbeddingGenerator<string, Embedding<float>> embeddings, DataContext db, IConfiguration config)
    {
        _chat = chat;
        _tools = tools;
        _history = history;
        _embeddings = embeddings;
        _db = db;
        _config = config;
    }

    [HttpPost("ask")]
    public async Task<IActionResult> Ask(AskRequest request)
    {
        var options = new ChatOptions
        {
            Temperature = request.Temperature,
            MaxOutputTokens = request.MaxOutputTokens
        };

        ChatResponse response =
            await _chat.GetResponseAsync(request.Prompt, options);
        return Ok(AskResponse.From(response));
    }

    [HttpPost("assistant")]
    public async Task<IActionResult> Assistant(AskRequest request)
    {
        List<ChatMessage> messages =
        [
            new(ChatRole.System, $"""
                You are the assistant of the restaurant "{RestaurantName}".
                You answer only about the menu, bookings and hours ({OpenHours}).
                Answer: at most 3 sentences, in the language the user writes in.
                On any other topic you politely decline and say what you can help with.
                You don't invent prices or ingredients — you say they are on the menu.
                """),
            new(ChatRole.User, request.Prompt)
        ];

        ChatResponse response = await _chat.GetResponseAsync(messages);
        return Ok(AskResponse.From(response));
    }

    public enum Sentiment { Positive, Negative, Mixed }

    [HttpPost("review-sentiment")]
    public async Task<IActionResult> ReviewSentiment(AskRequest request)
    {
        List<ChatMessage> messages =
        [
            new(ChatRole.System, "Classify the restaurant review."),
            new(ChatRole.User, request.Prompt)
        ];

        ChatResponse<Sentiment> response =
            await _chat.GetResponseAsync<Sentiment>(messages);

        Sentiment sentiment = response.Result;
        if (sentiment == Sentiment.Negative) { /* notify the manager */ }

        return Ok(sentiment.ToString());
    }

    public record BookingRequest(
        [property: Description("Booking date and time")] DateTime? At,
        [property: Description("Number of guests")] int? Guests);

    [HttpPost("parse-booking")]
    public async Task<IActionResult> ParseBooking(AskRequest request)
    {
        List<ChatMessage> messages =
        [
            new(ChatRole.System, $"""
                Extract the booking details from the user's text.
                Today is {DateTime.Now:yyyy-MM-dd, dddd}.
                Leave null whatever is not in the text.
                """),
            new(ChatRole.User, request.Prompt)
        ];

        ChatResponse<BookingRequest> response =
            await _chat.GetResponseAsync<BookingRequest>(messages);
        return Ok(response.Result);
    }

    public record ChatRequest(Guid? ConversationId, string Prompt);

    [HttpPost("chat")]
    public async Task<IActionResult> Chat(ChatRequest request, CancellationToken ct)
    {
        Guid conversationId = request.ConversationId ?? Guid.NewGuid();

        List<ChatMessage> history = await _history.LoadAsync(conversationId, ct);
        IEnumerable<ChatMessage> recent = await Reducer.ReduceAsync(history, ct);

        List<FoundChunk> sources = await SearchChunksAsync(request.Prompt, ct);   // retrieve
        ChatMessage question = new(ChatRole.User, request.Prompt);
        ChatMessage augmented = new(ChatRole.User, $"""
            Sources:
            {BuildContext(sources)}

            Question: {request.Prompt}
            """);                                                                  // augment

        List<ChatMessage> messages = [SystemPrompt(), .. recent, augmented];
        ChatResponse response =
            await _chat.GetResponseAsync(messages, ChatToolOptions(), ct);         // generate

        // into the history — the original question, not the one with the sources
        await _history.AppendAsync(conversationId, [question, .. response.Messages], ct);

        return Ok(new { conversationId, response.Text,
            Sources = sources.Select((s, i) => new { N = i + 1, s.File, s.Page }) });
    }

    [HttpPost("chat/chunks")]
    public async Task<IActionResult> ChatChunks(AskRequest request)
    {
        Stopwatch sw = Stopwatch.StartNew();
        List<object> chunks = [];

        await foreach (ChatResponseUpdate update in
            _chat.GetStreamingResponseAsync(request.Prompt))
        {
            chunks.Add(new { Ms = sw.ElapsedMilliseconds, update.Text });
        }

        return Ok(chunks);
    }

    [HttpGet("chat/stream")]
    public IResult ChatStream(string prompt, CancellationToken cancellationToken)
    {
        return TypedResults.ServerSentEvents(StreamAnswer(prompt, cancellationToken));
    }

    private async IAsyncEnumerable<string> StreamAnswer(string prompt,
        [EnumeratorCancellation] CancellationToken cancellationToken)
    {
        List<ChatMessage> messages = [SystemPrompt(), new(ChatRole.User, prompt)];

        await foreach (ChatResponseUpdate update in
            _chat.GetStreamingResponseAsync(messages, ChatToolOptions(), cancellationToken))
        {
            if (!string.IsNullOrEmpty(update.Text))
                yield return update.Text;
        }
    }

    [HttpPost("embed")]
    public async Task<IActionResult> Embed(AskRequest request, CancellationToken ct)
    {
        ReadOnlyMemory<float> vector =
            await _embeddings.GenerateVectorAsync(request.Prompt, cancellationToken: ct);

        return Ok(new { vector.Length, First5 = vector.ToArray()[..5] });
    }

    [HttpPost("menu/index")]
    public async Task<IActionResult> IndexMenu(CancellationToken ct)
    {
        string model = _config["Gemini:EmbeddingModel"]!;

        // only the dishes that have no vector, or have one from another model
        List<Product> products = await _db.Products
            .Where(p => p.Embedding == null || p.EmbeddingModel != model)
            .ToListAsync(ct);

        foreach (Product[] batch in products.Chunk(100))          // Gemini: up to 100 per request
        {
            var vectors = await _embeddings.GenerateAsync(
                batch.Select(p => EmbeddingText.Document(p.Name, p.Description)),
                cancellationToken: ct);

            for (int i = 0; i < batch.Length; i++)
            {
                batch[i].Embedding = new Vector(vectors[i].Vector);  // ReadOnlyMemory<float> → Vector
                batch[i].EmbeddingModel = model;
            }
        }

        await _db.SaveChangesAsync(ct);
        return Ok(new { Indexed = products.Count });
    }

    [HttpGet("menu/search")]
    public async Task<IActionResult> SearchMenu(string q, CancellationToken ct)
    {
        var query = new Vector(await _embeddings.GenerateVectorAsync(
            EmbeddingText.Query(q), cancellationToken: ct));

        var top = await _db.Products
            .Where(p => p.Embedding != null)
            .OrderBy(p => p.Embedding!.CosineDistance(query))       // → ORDER BY "Embedding" <=> @query
            .Select(p => new { p.Name, Distance = p.Embedding!.CosineDistance(query) })
            .Take(3)                                                  // → LIMIT 3
            .ToListAsync(ct);

        return Ok(top.Select(x => new { x.Name, Score = 1 - x.Distance }));   // distance → similarity
    }

    [HttpPost("documents")]
    public async Task<IActionResult> UploadDocument(IFormFile file, CancellationToken ct)
    {
        using var ms = new MemoryStream();
        await file.CopyToAsync(ms, ct);

        // 1–2: PDF → pages → parts
        var doc = new Document { FileName = file.FileName, UploadedAt = DateTime.UtcNow };
        foreach (var (page, text) in PdfText.ReadPages(ms.ToArray()))
            foreach (string chunk in TextChunker.Split(text))
                doc.Chunks.Add(new DocumentChunk { Page = page, Content = chunk });

        // 3: vectors, in batches of 100
        string model = _config["Gemini:EmbeddingModel"]!;
        foreach (DocumentChunk[] batch in doc.Chunks.Chunk(100))
        {
            var vectors = await _embeddings.GenerateAsync(
                batch.Select(c => EmbeddingText.Document(doc.FileName, c.Content)),
                cancellationToken: ct);
            for (int i = 0; i < batch.Length; i++)
            {
                batch[i].Embedding = new Vector(vectors[i].Vector);
                batch[i].EmbeddingModel = model;
            }
        }

        // 4: into the database — the document and all its parts together
        _db.Documents.Add(doc);
        await _db.SaveChangesAsync(ct);
        return Ok(new { doc.Id, Chunks = doc.Chunks.Count });
    }

    public record FoundChunk(string File, int Page, string Content, double Score);

    private async Task<List<FoundChunk>> SearchChunksAsync(string q, CancellationToken ct)
    {
        float minScore = _config.GetValue<float>("Rag:MinScore");   // by testing — like lecture 9's homework
        var query = new Vector(await _embeddings.GenerateVectorAsync(
            EmbeddingText.Query(q), cancellationToken: ct));

        List<FoundChunk> top = await _db.DocumentChunks
            .Where(c => c.Embedding != null)
            .OrderBy(c => c.Embedding!.CosineDistance(query))
            .Select(c => new FoundChunk(c.Document.FileName, c.Page, c.Content,
                1 - c.Embedding!.CosineDistance(query)))
            .Take(5)
            .ToListAsync(ct);

        return top.Where(c => c.Score >= minScore).ToList();
    }

    private static string BuildContext(List<FoundChunk> chunks) =>
        chunks.Count == 0
            ? "(No relevant information was found in the sources)"
            : string.Join("\n\n", chunks.Select((c, i) => $"[{i + 1}] ({c.File}, p. {c.Page})\n{c.Content}"));

    // not stored in the database: it is ours and is built anew every time (with today's date)
    private static ChatMessage SystemPrompt() => new(ChatRole.System, $"""
        You are a restaurant assistant. Answer briefly.
        Take menu and free-table data only from the tools, don't make anything up.
        On the restaurant's rules answer only from the sources and give the number: [1].
        If it isn't in the sources — say: "I don't have information on this".
        Today is {DateTime.Now:yyyy-MM-dd, dddd}.
        """);

    private ChatOptions ChatToolOptions() => new()
    {
        Tools =
        [
            AIFunctionFactory.Create(_tools.GetMenu, name: "get_menu"),
            AIFunctionFactory.Create(_tools.CheckAvailability, name: "check_availability")
        ]
    };
}
```

```csharp Controllers/AiLabController.cs · ChatStream (RAG) — ⭐
private async IAsyncEnumerable<SseItem<string>> StreamRag(string prompt,
    [EnumeratorCancellation] CancellationToken ct)
{
    List<FoundChunk> sources = await SearchChunksAsync(prompt, ct);
    yield return new SseItem<string>(
        JsonSerializer.Serialize(sources.Select((s, i) => new { N = i + 1, s.File, s.Page })),
        eventType: "sources");

    List<ChatMessage> messages = [SystemPrompt(), new(ChatRole.User, $"""
        Sources:
        {BuildContext(sources)}

        Question: {prompt}
        """)];
    await foreach (ChatResponseUpdate update in _chat.GetStreamingResponseAsync(messages, ChatToolOptions(), ct))
        if (!string.IsNullOrEmpty(update.Text))
            yield return new SseItem<string>(update.Text);
}
```
