// _search: DocumentSearch from lecture 11 [HttpPost("retrieval")] public async Task Retrieval(CancellationToken ct) { string json = await System.IO.File.ReadAllTextAsync("golden.json", ct); var golden = JsonSerializer.Deserialize>(json, JsonSerializerOptions.Web)!; int hits = 0; double reciprocalRanks = 0; List misses = []; foreach (GoldenItem item in golden) { List found = await _search.SearchAsync(item.Question, ct); int rank = found.FindIndex(c => c.Content.Contains(item.MustContain)) + 1; if (rank is >= 1 and <= 3) hits++; else misses.Add(item.Question); if (rank > 0) reciprocalRanks += 1.0 / rank; } return Ok(new RetrievalReport((double)hits / golden.Count, reciprocalRanks / golden.Count, misses)); }

var queryVector = new Vector(await _embeddings.GenerateVectorAsync( EmbeddingInput.Query(query), cancellationToken: ct)); var words = EF.Functions.ToTsQuery("english", OrQuery(query)); List byMeaning = await _db.Chunks .Where(c => 1 - c.Embedding!.CosineDistance(queryVector) >= _minScore) .OrderBy(c => c.Embedding!.CosineDistance(queryVector)) .Select(c => c.Id).Take(20).ToListAsync(ct); List byWords = await _db.Chunks .Where(c => c.SearchVector.Matches(words)) .OrderByDescending(c => c.SearchVector.Rank(words)) .Select(c => c.Id).Take(20).ToListAsync(ct); List top = Rrf(byMeaning, byWords).Take(5).ToList(); static IEnumerable Rrf(params List[] lists) => lists .SelectMany(list => list.Select((id, i) => (id, score: 1.0 / (60 + i + 1)))) .GroupBy(x => x.id) .OrderByDescending(g => g.Sum(x => x.score)) .Select(g => g.Key);
modelBuilder.Entity() .HasGeneratedTsVectorColumn( c => c.SearchVector, "english", c => new { c.Content }) .HasIndex(c => c.SearchVector) .HasMethod("GIN");
