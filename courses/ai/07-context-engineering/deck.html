public class ChatHistoryStore { // jsonb reorders keys — "$type" is no longer first private static readonly JsonSerializerOptions Json = new(AIJsonUtilities.DefaultOptions) { AllowOutOfOrderMetadataProperties = true }; private readonly DataContext _db; public ChatHistoryStore(DataContext db) { _db = db; } public async Task> LoadAsync(Guid conversationId, CancellationToken ct) { List rows = await _db.ChatMessages .Where(m => m.ConversationId == conversationId) .OrderBy(m => m.Position) .Select(m => m.Json) .ToListAsync(ct); return rows.Select(j => JsonSerializer.Deserialize(j, Json)!).ToList(); } public async Task AppendAsync(Guid conversationId, IEnumerable messages, CancellationToken ct) { int position = await _db.ChatMessages.CountAsync(m => m.ConversationId == conversationId, ct); foreach (ChatMessage message in messages) { _db.ChatMessages.Add(new ChatMessageRecord { Id = Guid.NewGuid(), ConversationId = conversationId, Position = position++, Role = message.Role.Value, Json = JsonSerializer.Serialize(message, Json), CreatedAt = DateTime.UtcNow }); } await _db.SaveChangesAsync(ct); } }
