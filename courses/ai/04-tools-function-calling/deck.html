public record AvailabilityInfo(bool Available, int FreeTables); public sealed class TableTools(DataContext db) { [Description("Checks whether a table is free at the given time")] public async Task CheckAvailability( [Description("Booking date and time")] DateTime at, [Description("Number of guests")] int guests ) { var from = at.AddHours(-2); var to = at.AddHours(2); var free = await db.Tables.CountAsync(t => t.Seats >= guests && !db.Bookings.Any(b => b.TableId == t.Id && b.At > from && b.At < to) ); return new AvailabilityInfo(free > 0, free); } }
Tools = [ AIFunctionFactory.Create(menu.GetMenu, name: "get_menu"), AIFunctionFactory.Create(table.CheckAvailability, name: "check_availability") ]
