using UglyToad.PdfPig; using UglyToad.PdfPig.DocumentLayoutAnalysis.TextExtractor; public static class PdfText { public static List<(int Page, string Text)> ReadPages(byte[] pdf) { using PdfDocument doc = PdfDocument.Open(pdf); return doc.GetPages() .Select(p => (p.Number, Clean(ContentOrderTextExtractor.GetText(p)))) .Where(p => p.Item2.Length > 0) .ToList(); } private static string Clean(string text) => Regex.Replace(text, @"\s+", " ").Trim(); }

private async Task> SearchChunksAsync(string q, CancellationToken ct) { float minScore = _config.GetValue("Rag:MinScore"); var query = new Vector(await _embeddings.GenerateVectorAsync( EmbeddingText.Query(q), cancellationToken: ct)); List top = await _db.DocumentChunks .Where(c => c.Embedding != null) .OrderBy(c => c.Embedding!.CosineDistance(query)) .Select(c => new FoundChunk(c.Document.FileName, c.Page, c.Content, 1 - c.Embedding!.CosineDistance(query))) .Take(5) .ToListAsync(ct); return top.Where(c => c.Score >= minScore).ToList(); }
