static float Cosine(ReadOnlySpan a, ReadOnlySpan b) { float dot = 0, na = 0, nb = 0; for (int i = 0; i < a.Length; i++) { dot += a[i] * b[i]; na += a[i] * a[i]; nb += b[i] * b[i]; } return dot / (MathF.Sqrt(na) * MathF.Sqrt(nb)); }
float score = TensorPrimitives.CosineSimilarity(a.Span, b.Span);
