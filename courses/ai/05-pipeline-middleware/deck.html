using Microsoft.Extensions.AI; public class TokenCounterChatClient : DelegatingChatClient { private readonly ILogger _logger; public TokenCounterChatClient(IChatClient inner, ILogger logger) : base(inner) { _logger = logger; } public override async Task GetResponseAsync( IEnumerable messages, ChatOptions? options = null, CancellationToken cancellationToken = default) { // ↓ here is the request — it hasn't reached Gemini yet ChatResponse response = await base.GetResponseAsync(messages, options, cancellationToken); // ↑ here is the response — Gemini has already answered _logger.LogInformation("🔢 in: {In}, out: {Out}, total: {Total}", response.Usage?.InputTokenCount, response.Usage?.OutputTokenCount, response.Usage?.TotalTokenCount); return response; } }

public class InputGuardChatClient : DelegatingChatClient { private const int MaxLength = 1000; public InputGuardChatClient(IChatClient inner) : base(inner) { } public override Task GetResponseAsync( IEnumerable messages, ChatOptions? options = null, CancellationToken cancellationToken = default) { string? question = messages.LastOrDefault(m => m.Role == ChatRole.User)?.Text; if (question is not null && question.Length > MaxLength) { ChatMessage answer = new(ChatRole.Assistant, $"The question is too long — {MaxLength} characters at most."); return Task.FromResult(new ChatResponse(answer)); } return base.GetResponseAsync(messages, options, cancellationToken); } }
