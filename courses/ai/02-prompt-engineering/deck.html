[HttpPost("ask")] public async Task Ask(PromptRequest request) { var options = new ChatOptions { Temperature = request.Temperature, MaxOutputTokens = request.MaxOutputTokens }; ChatResponse response = await _chat.GetResponseAsync(request.Prompt, options); return Ok(PromptResponse.From(response)); }
public record PromptRequest( string Prompt, float? Temperature = null, int? MaxOutputTokens = null);
